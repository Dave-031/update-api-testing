@echo off
setlocal EnableExtensions EnableDelayedExpansion
:: ============================================================================
::  APP UPDATER  -  step-by-step updater with self-update support
:: ============================================================================

:: ------------------------------- SETTINGS -----------------------------------
:: Change these to configure the updater.
set "UPDATER_VERSION=unknown"
:: (Version above is only a fallback. The real version is detected
::  automatically by comparing this file's MD5 with the server.)
set "APP_NAME=MyApp"
set "API_URL=https://api.cc1.dpdns.org/update.php"
:: Folder where downloaded zip files are kept
set "DOWNLOAD_DIR=%~dp0downloads"
:: Folder the app gets extracted into
set "INSTALL_DIR=%~dp0app"
:: File that stores the currently installed version + commit
set "VERSION_FILE=%~dp0installed_version.txt"
:: 1 = check for updater updates first, 0 = skip
set "SELF_UPDATE=1"
:: 1 = extract the zip into INSTALL_DIR after download, 0 = only download
set "AUTO_EXTRACT=1"
:: 1 = wipe INSTALL_DIR before extracting, 0 = overwrite files in place
set "CLEAN_INSTALL=0"
:: 1 = delete older zip files after a successful update
set "DELETE_OLD_ZIPS=1"
:: How many times to retry a failed download
set "MAX_RETRIES=3"
:: Seconds to wait between retries
set "RETRY_DELAY=3"
:: 1 = pause at the end so the window stays open
set "PAUSE_AT_END=1"
:: 1 = enable colored text, 0 = plain text
set "USE_COLOR=1"

:: ------------------------------- COLORS -------------------------------------
:: ANSI color codes (30-37 normal, 90-97 bright). Change to taste.
set "C_TITLE=96"
set "C_STEP=95"
set "C_INFO=97"
set "C_OK=92"
set "C_WARN=93"
set "C_ERR=91"
set "C_DIM=90"
set "C_VAL=94"
:: ============================================================================

title %APP_NAME% Updater
for /f %%a in ('echo prompt $E^| cmd') do set "ESC=%%a"
if "%USE_COLOR%"=="1" (
    set "T=%ESC%[%C_TITLE%m" & set "S=%ESC%[%C_STEP%m" & set "I=%ESC%[%C_INFO%m"
    set "G=%ESC%[%C_OK%m"    & set "W=%ESC%[%C_WARN%m" & set "E=%ESC%[%C_ERR%m"
    set "D=%ESC%[%C_DIM%m"   & set "V=%ESC%[%C_VAL%m"  & set "R=%ESC%[0m"
) else (
    set "T=" & set "S=" & set "I=" & set "G=" & set "W=" & set "E=" & set "D=" & set "V=" & set "R="
)

set "TOTAL_STEPS=7"
set "STEP=0"
set "TMP_DIR=%TEMP%\%APP_NAME%_updater"
set "API_JSON=%TMP_DIR%\api.json"
set "API_VARS=%TMP_DIR%\api_vars.txt"
if not exist "%TMP_DIR%" mkdir "%TMP_DIR%"

echo.
echo %T%=================================================================%R%
echo %T%   %APP_NAME% Updater%R%
echo %T%=================================================================%R%
set "JUST_UPDATED=0"
if /i "%~1"=="--updated" (
    set "JUST_UPDATED=1"
    if not "%~2"=="" set "UPDATER_VERSION=%~2"
    echo %G%  Updater was updated to v%~2 - continuing update...%R%
)
echo.

:: ---------------------------------------------------------------------------
call :step "Checking required tools"
where powershell >nul 2>&1 || (call :fail "PowerShell was not found. It is required." & goto :end)
call :ok "PowerShell found"
where certutil >nul 2>&1 && call :ok "certutil found" || call :warn "certutil not found (not required)"

:: ---------------------------------------------------------------------------
call :step "Contacting update server"
call :info "API: %V%%API_URL%"
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "try { [Net.ServicePointManager]::SecurityProtocol=[Net.SecurityProtocolType]::Tls12; $j = Invoke-RestMethod -Uri '%API_URL%' -UseBasicParsing -TimeoutSec 20;" ^
  "$o = @(('COMMIT_ID=' + $j.commit_id), ('APP_VERSION=' + $j.app_version), ('DOWNLOAD_URL=' + $j.download_url)," ^
  "('FILE_SIZE=' + $j.size), ('FILE_MD5=' + ([string]$j.md5).ToLower()), ('UPD_VERSION=' + $j.updater.version)," ^
  "('UPD_URL=' + $j.updater.download_url), ('UPD_MD5=' + ([string]$j.updater.md5).ToLower()), ('API_MESSAGE=' + $j.message));" ^
  "$o | Set-Content -Encoding ASCII '%API_VARS%'; exit 0 } catch { Write-Host $_.Exception.Message; exit 1 }"
if errorlevel 1 (call :fail "Could not reach the update server." & goto :end)
for /f "usebackq tokens=1,* delims==" %%a in ("%API_VARS%") do set "%%a=%%b"
for %%k in (COMMIT_ID APP_VERSION DOWNLOAD_URL FILE_MD5 UPD_URL UPD_MD5) do if not defined %%k (call :fail "Server response is missing %%k." & goto :end)
set "SHORT_COMMIT=%COMMIT_ID:~0,6%"
call :ok "Server says: %V%%API_MESSAGE%"
call :info "Latest app version : %V%%APP_VERSION%%I%  (commit %V%%SHORT_COMMIT%%I%)"
call :info "Download size      : %V%%FILE_SIZE%%I% bytes"
call :info "Latest updater     : %V%%UPD_VERSION%"

:: ---------------------------------------------------------------------------
call :step "Checking for updater updates"
if "%SELF_UPDATE%"=="0" (call :info "Skipped." & goto :after_self)
call :md5 "%~f0"
set "SELF_MD5=!HASH!"
call :info "This updater MD5 : %V%!SELF_MD5!"
call :info "Latest updater MD5: %V%%UPD_MD5%"
if /i "!SELF_MD5!"=="%UPD_MD5%" (
    set "UPDATER_VERSION=%UPD_VERSION%"
    title %APP_NAME% Updater v%UPD_VERSION%
    call :ok "Updater is the latest version: v%UPD_VERSION% (MD5 match)"
    goto :after_self
)
if "!JUST_UPDATED!"=="1" (
    call :warn "Just updated but MD5 still differs - skipping to avoid a loop."
    call :warn "Check that the MD5 in the API matches the uploaded updater file."
    goto :after_self
)
call :warn "Updater is out of date (MD5 differs). Getting v%UPD_VERSION%..."
set "NEW_UPDATER=%TMP_DIR%\updater-%UPD_VERSION%-%SHORT_COMMIT%.bat"
call :download "%UPD_URL%" "%NEW_UPDATER%"
if errorlevel 1 (call :warn "Updater download failed, continuing with current version." & goto :after_self)
call :md5 "%NEW_UPDATER%"
if /i not "!HASH!"=="%UPD_MD5%" (
    call :warn "Updater MD5 mismatch (got !HASH!, expected %UPD_MD5%). Not replacing."
    goto :after_self
)
call :ok "New updater verified (MD5 %UPD_MD5%)"
call :info "Replacing myself and restarting..."
set "SWAP=%TMP_DIR%\swap.cmd"
> "%SWAP%" (
    echo @echo off
    echo timeout /t 2 /nobreak ^>nul
    echo copy /y "%NEW_UPDATER%" "%~f0" ^>nul
    echo start "" "%~f0" --updated %UPD_VERSION%
    echo del "%%~f0"
)
start "" /min cmd /c "%SWAP%"
exit /b 0
:after_self

:: ---------------------------------------------------------------------------
call :step "Checking your current version"
set "INSTALLED_VERSION=none"
set "INSTALLED_COMMIT=none"
if exist "%VERSION_FILE%" (
    for /f "usebackq tokens=1,* delims==" %%a in ("%VERSION_FILE%") do set "INSTALLED_%%a=%%b"
)
call :info "Installed version: %V%!INSTALLED_VERSION!%I%  (commit %V%!INSTALLED_COMMIT!%I%)"
set "ZIP_NAME=%APP_NAME%-%APP_VERSION%-%SHORT_COMMIT%.zip"
set "ZIP_PATH=%DOWNLOAD_DIR%\%ZIP_NAME%"
if not exist "%DOWNLOAD_DIR%" mkdir "%DOWNLOAD_DIR%"
set "NEED_DOWNLOAD=1"
if exist "%ZIP_PATH%" (
    call :info "Found existing file %V%%ZIP_NAME%%I%, checking MD5..."
    call :md5 "%ZIP_PATH%"
    if /i "!HASH!"=="%FILE_MD5%" (
        call :ok "Existing download matches the latest file."
        set "NEED_DOWNLOAD=0"
    ) else (
        call :warn "MD5 mismatch (!HASH!), file will be re-downloaded."
        del /q "%ZIP_PATH%" >nul 2>&1
    )
) else (
    call :info "Latest file not downloaded yet."
)

:: ---------------------------------------------------------------------------
call :step "Downloading update"
if "!NEED_DOWNLOAD!"=="0" (call :ok "Nothing to download." & goto :after_dl)
set /a TRY=0
:dl_retry
set /a TRY+=1
call :info "Attempt !TRY!/%MAX_RETRIES%: %V%%ZIP_NAME%"
call :download "%DOWNLOAD_URL%" "%ZIP_PATH%"
if not errorlevel 1 (
    for %%f in ("%ZIP_PATH%") do set "GOT_SIZE=%%~zf"
    call :md5 "%ZIP_PATH%"
    if /i "!HASH!"=="%FILE_MD5%" (
        call :ok "Downloaded !GOT_SIZE! bytes, MD5 verified."
        goto :after_dl
    )
    call :warn "Verification failed (size !GOT_SIZE!/%FILE_SIZE%, MD5 !HASH!)."
    del /q "%ZIP_PATH%" >nul 2>&1
)
if !TRY! LSS %MAX_RETRIES% (
    call :warn "Retrying in %RETRY_DELAY% seconds..."
    timeout /t %RETRY_DELAY% /nobreak >nul
    goto :dl_retry
)
call :fail "Download failed after %MAX_RETRIES% attempts."
goto :end
:after_dl

:: ---------------------------------------------------------------------------
call :step "Installing update"
if "%AUTO_EXTRACT%"=="0" (call :info "Auto-extract disabled. File saved to %V%%ZIP_PATH%" & goto :after_install)
if /i "!INSTALLED_COMMIT!"=="%COMMIT_ID%" if exist "%INSTALL_DIR%" (
    call :ok "Version %APP_VERSION% is already installed."
    goto :after_install
)
if "%CLEAN_INSTALL%"=="1" if exist "%INSTALL_DIR%" (
    call :info "Clearing old install folder..."
    rmdir /s /q "%INSTALL_DIR%"
)
if not exist "%INSTALL_DIR%" mkdir "%INSTALL_DIR%"
call :info "Extracting to %V%%INSTALL_DIR%"
powershell -NoProfile -Command "try { Expand-Archive -LiteralPath '%ZIP_PATH%' -DestinationPath '%INSTALL_DIR%' -Force; exit 0 } catch { Write-Host $_.Exception.Message; exit 1 }"
if errorlevel 1 (call :fail "Extraction failed." & goto :end)
> "%VERSION_FILE%" (
    echo VERSION=%APP_VERSION%
    echo COMMIT=%COMMIT_ID%
)
call :ok "Installed %APP_NAME% v%APP_VERSION% (%SHORT_COMMIT%)"
:after_install

:: ---------------------------------------------------------------------------
call :step "Cleaning up"
if "%DELETE_OLD_ZIPS%"=="1" (
    for %%f in ("%DOWNLOAD_DIR%\%APP_NAME%-*.zip") do (
        if /i not "%%~nxf"=="%ZIP_NAME%" (
            del /q "%%f" >nul 2>&1
            call :info "Removed old file %D%%%~nxf"
        )
    )
)
rmdir /s /q "%TMP_DIR%" >nul 2>&1
call :ok "Done."

echo.
echo %G%=================================================================%R%
echo %G%   Update complete!  %APP_NAME% is on v%APP_VERSION% (%SHORT_COMMIT%)%R%
echo %G%=================================================================%R%
goto :end

:: =============================== HELPERS ====================================
:step
set /a STEP+=1
echo.
echo %S%[Step %STEP%/%TOTAL_STEPS%]%R% %I%%~1...%R%
exit /b 0

:info
echo    %D%-%R% %I%%~1%R%
exit /b 0

:ok
echo    %G%[OK]%R% %~1
exit /b 0

:warn
echo    %W%[WARN]%R% %W%%~1%R%
exit /b 0

:fail
echo    %E%[FAIL] %~1%R%
exit /b 0

:: :download <url> <outfile>   -> errorlevel 0 on success
:download
powershell -NoProfile -Command "try { [Net.ServicePointManager]::SecurityProtocol=[Net.SecurityProtocolType]::Tls12; $ProgressPreference='SilentlyContinue'; (New-Object Net.WebClient).DownloadFile('%~1','%~2'); exit 0 } catch { Write-Host $_.Exception.Message; exit 1 }"
exit /b %errorlevel%

:: :md5 <file>  -> sets HASH (lowercase)
:md5
set "HASH="
for /f "usebackq delims=" %%h in (`powershell -NoProfile -Command "(Get-FileHash -Algorithm MD5 -LiteralPath '%~1').Hash.ToLower()"`) do set "HASH=%%h"
exit /b 0

:: :vercmp <a> <b>  -> VERCMP = 1 if a>b, 0 if equal, -1 if a<b
:vercmp
for /f "usebackq delims=" %%r in (`powershell -NoProfile -Command "([version]'%~1').CompareTo([version]'%~2')"`) do set "VERCMP=%%r"
exit /b 0

:end
echo.
if "%PAUSE_AT_END%"=="1" pause
endlocal
exit /b 0
