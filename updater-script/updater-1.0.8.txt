@echo off
setlocal enabledelayedexpansion

cls
color 0F

REM ============================================================
REM Application Updater Script v1.0.0
REM Auto-update with MD5 verification and smart updater swap
REM ============================================================

REM API Configuration
set "API_URL=https://api.cc1.dpdns.org/update.php"
set "SCRIPT_DIR=%~dp0"
set "SCRIPT_NAME=%~nx0"
set "TEMP_JSON=%TEMP%\update-info.json"
set "UPDATER_HASH_FILE=%SCRIPT_DIR%updater.hash"
set "APP_HASH_FILE=%SCRIPT_DIR%app.hash"

REM Color codes
set "COLOR_GREEN=0A"
set "COLOR_RED=0C"
set "COLOR_YELLOW=0E"
set "COLOR_BLUE=09"
set "COLOR_WHITE=0F"
set "COLOR_GRAY=08"

REM Step tracking
set "STEP=0"

call :print_header

REM ============================================================
REM STEP 1: Fetch API Data
REM ============================================================
call :increment_step
call :print_step "Fetching update information from API"
call :fetch_api_data

if errorlevel 1 (
    call :print_error "Failed to fetch API data"
    pause
    exit /b 1
)

REM ============================================================
REM STEP 2: Parse JSON Response
REM ============================================================
call :increment_step
call :print_step "Parsing update information"
call :parse_json

if errorlevel 1 (
    call :print_error "Failed to parse JSON data"
    pause
    exit /b 1
)

REM ============================================================
REM STEP 3: Calculate Current Updater MD5
REM ============================================================
call :increment_step
call :print_step "Checking updater version"
call :get_updater_md5

call :print_info "   Current Updater MD5: !CURRENT_UPDATER_MD5!"
call :print_info "   Latest Updater MD5: !LATEST_UPDATER_MD5!"

REM ============================================================
REM STEP 4: Download and Update Updater (if needed)
REM ============================================================
if not "!CURRENT_UPDATER_MD5!"=="!LATEST_UPDATER_MD5!" (
    call :increment_step
    call :print_step "Downloading updater version !LATEST_UPDATER_VERSION!"
    call :download_updater
    
    if errorlevel 1 (
        call :print_error "Failed to download updater"
        pause
        exit /b 1
    )
    
    call :increment_step
    call :print_step "Installing updater update"
    call :install_updater_update
    
    call :print_success "Updater successfully updated to !LATEST_UPDATER_VERSION!"
) else (
    call :print_status "Updater is up to date"
)

REM ============================================================
REM STEP 5: Check App Version
REM ============================================================
call :increment_step
call :print_step "Checking application version"
call :get_app_md5

call :print_info "   Current App MD5: !CURRENT_APP_MD5!"
call :print_info "   Latest App MD5: !LATEST_APP_MD5!"

REM ============================================================
REM STEP 6: Download Application (if needed)
REM ============================================================
if not "!CURRENT_APP_MD5!"=="!LATEST_APP_MD5!" (
    call :increment_step
    call :print_step "Downloading application version !APP_VERSION! [Commit: !COMMIT_SHORT!]"
    call :download_app
    
    if errorlevel 1 (
        call :print_error "Failed to download application"
        pause
        exit /b 1
    )
    
    call :increment_step
    call :print_step "Verifying download integrity"
    call :verify_download
    
    if errorlevel 1 (
        call :print_error "Download verification failed"
        pause
        exit /b 1
    )
    
    call :save_app_hash
    call :print_success "Application updated to !APP_VERSION!"
) else (
    call :print_status "Application is up to date [!APP_VERSION!]"
)

REM ============================================================
REM Completion
REM ============================================================
call :print_success "Update process completed successfully!"
call :print_info "Application: !APP_VERSION! [!COMMIT_SHORT!]"
call :print_info "Updater: !LATEST_UPDATER_VERSION!"
echo.
timeout /t 3 /nobreak
exit /b 0

REM ============================================================
REM SUBROUTINES
REM ============================================================

:print_header
color 0F
cls
echo.
color 0E
echo ====================================================
echo  APPLICATION AUTO UPDATER v1.0.0
echo  Automatic Update System Active
echo ====================================================
color 0F
echo.
exit /b 0

:increment_step
set /a STEP+=1
exit /b 0

:print_step
color 09
echo [STEP %STEP%] %~1
color 0F
exit /b 0

:print_status
color 0B
echo [*] %~1
color 0F
exit /b 0

:print_success
color 0A
echo [+] %~1
color 0F
exit /b 0

:print_error
color 0C
echo [-] %~1
color 0F
exit /b 0

:print_info
color 08
echo %~1
color 0F
exit /b 0

:fetch_api_data
echo   Connecting to !API_URL!...
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "$response = Invoke-WebRequest -Uri '!API_URL!' -UseBasicParsing -ErrorAction Stop; $response.Content | Out-File '!TEMP_JSON!' -Encoding UTF8" 2>nul

if errorlevel 1 (
    exit /b 1
)

echo   Connection successful
exit /b 0

:parse_json
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "($json = Get-Content '!TEMP_JSON!' ^| ConvertFrom-Json); Write-Host $json.commit_id"') do set "COMMIT_ID=%%A"
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "($json = Get-Content '!TEMP_JSON!' ^| ConvertFrom-Json); Write-Host $json.app_version"') do set "APP_VERSION=%%A"
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "($json = Get-Content '!TEMP_JSON!' ^| ConvertFrom-Json); Write-Host $json.download_url"') do set "APP_URL=%%A"
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "($json = Get-Content '!TEMP_JSON!' ^| ConvertFrom-Json); Write-Host $json.updater.version"') do set "LATEST_UPDATER_VERSION=%%A"
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "($json = Get-Content '!TEMP_JSON!' ^| ConvertFrom-Json); Write-Host $json.updater.download_url"') do set "UPDATER_URL=%%A"

REM Get first 6 characters of commit ID
set "COMMIT_SHORT=!COMMIT_ID:~0,6!"

REM Calculate MD5 of latest versions from URL
call :calculate_url_md5 "!APP_URL!" LATEST_APP_MD5
call :calculate_url_md5 "!UPDATER_URL!" LATEST_UPDATER_MD5

echo   Commit ID: !COMMIT_SHORT!
echo   App Version: !APP_VERSION!
echo   Updater Version: !LATEST_UPDATER_VERSION!

exit /b 0

:get_updater_md5
REM Calculate MD5 of current running script
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "(Get-FileHash '!SCRIPT_DIR!!SCRIPT_NAME!' -Algorithm MD5).Hash"') do set "CURRENT_UPDATER_MD5=%%A"

if not defined CURRENT_UPDATER_MD5 (
    set "CURRENT_UPDATER_MD5=NONE"
)

exit /b 0

:get_app_md5
REM Check if app hash file exists
if exist "!APP_HASH_FILE!" (
    for /f "tokens=*" %%A in ('type "!APP_HASH_FILE!"') do set "CURRENT_APP_MD5=%%A"
) else (
    set "CURRENT_APP_MD5=NONE"
)

exit /b 0

:calculate_url_md5
REM Download temp file and calculate its MD5
setlocal enabledelayedexpansion
set "URL=%~1"
set "VAR_NAME=%~2"
set "TEMP_FILE=%TEMP%\temp-download-%RANDOM%.tmp"

powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "$ProgressPreference = 'SilentlyContinue'; Invoke-WebRequest -Uri '%URL%' -OutFile '%TEMP_FILE%' -UseBasicParsing" 2>nul

if errorlevel 1 (
    endlocal
    exit /b 1
)

REM Wait a bit to ensure file is written
timeout /t 1 /nobreak >nul

for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "(Get-FileHash '%TEMP_FILE%' -Algorithm MD5).Hash"') do set "HASH=%%A"

del /f /q "!TEMP_FILE!" 2>nul

endlocal & set "%VAR_NAME%=%HASH%"
exit /b 0

:download_updater
set "NEW_UPDATER=%SCRIPT_DIR%updater-new-!LATEST_UPDATER_VERSION!.cmd"
echo   Downloading from: !UPDATER_URL!
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "$ProgressPreference = 'SilentlyContinue'; Invoke-WebRequest -Uri '!UPDATER_URL!' -OutFile '!NEW_UPDATER!' -UseBasicParsing" 2>nul

if errorlevel 1 (
    exit /b 1
)

REM Wait for file to be written
timeout /t 1 /nobreak >nul

echo   Download complete
exit /b 0

:install_updater_update
REM Backup old updater
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "(Get-FileHash '!SCRIPT_DIR!!SCRIPT_NAME!' -Algorithm MD5).Hash"') do set "OLD_HASH=%%A"

echo   Creating backup: updater-backup-!OLD_HASH:~0,8!.cmd
copy /Y "!SCRIPT_DIR!!SCRIPT_NAME!" "!SCRIPT_DIR!updater-backup-!OLD_HASH:~0,8!.cmd" >nul 2>&1

REM Replace current updater with new one
echo   Replacing current updater...
timeout /t 1 /nobreak >nul

REM Copy new updater to current location
copy /Y "!NEW_UPDATER!" "!SCRIPT_DIR!!SCRIPT_NAME!" >nul 2>&1

REM Delete temporary new updater file
del /f /q "!NEW_UPDATER!" 2>nul

echo   Updater switched successfully

exit /b 0

:download_app
set "APP_FILE=%SCRIPT_DIR%app-!APP_VERSION!-!COMMIT_SHORT!.zip"
echo   Downloading from CDN...
echo   Target: !APP_FILE!
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "$ProgressPreference = 'SilentlyContinue'; Invoke-WebRequest -Uri '!APP_URL!' -OutFile '!APP_FILE!' -UseBasicParsing" 2>nul

if errorlevel 1 (
    exit /b 1
)

REM Wait for file to be written
timeout /t 2 /nobreak >nul

echo   Download complete
exit /b 0

:verify_download
if not exist "!APP_FILE!" (
    call :print_error "Application file not found"
    exit /b 1
)

REM Simple file size check first
for %%F in ("!APP_FILE!") do set "FILE_SIZE=%%~zF"

if %FILE_SIZE% lss 1000 (
    call :print_error "Downloaded file is too small (%FILE_SIZE% bytes)"
    del /f /q "!APP_FILE!" 2>nul
    exit /b 1
)

echo   File size: %FILE_SIZE% bytes

REM Verify ZIP integrity with better error handling
powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "try { " ^
  "  [System.Reflection.Assembly]::LoadFrom([System.IO.Path]::Combine([System.Environment]::SystemDirectory, 'System.IO.Compression.dll')); " ^
  "  [System.IO.Compression.ZipFile]::OpenRead('!APP_FILE!').Dispose(); " ^
  "  Write-Host 'ZIP_VALID'; " ^
  "} catch { " ^
  "  Write-Host 'ZIP_INVALID'; " ^
  "  exit 1 " ^
  "}" 2>nul >"%TEMP%\zip_check.txt"

if errorlevel 1 (
    call :print_error "ZIP file verification failed or corrupted"
    del /f /q "!APP_FILE!" 2>nul
    del /f /q "%TEMP%\zip_check.txt" 2>nul
    exit /b 1
)

for /f "tokens=*" %%A in ('type "%TEMP%\zip_check.txt"') do set "ZIP_RESULT=%%A"
del /f /q "%TEMP%\zip_check.txt" 2>nul

if not "!ZIP_RESULT!"=="ZIP_VALID" (
    call :print_error "ZIP file is corrupted"
    del /f /q "!APP_FILE!" 2>nul
    exit /b 1
)

echo   Archive integrity verified
exit /b 0

:save_app_hash
for /f "tokens=*" %%A in ('powershell -NoProfile -ExecutionPolicy Bypass -Command "(Get-FileHash '!APP_FILE!' -Algorithm MD5).Hash"') do set "APP_HASH=%%A"

(
echo !APP_HASH!
) > "!APP_HASH_FILE!"

exit /b 0
